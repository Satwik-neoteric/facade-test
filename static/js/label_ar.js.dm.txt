        function fitImageToContainer() {
            const img = document.getElementById('main-image');
            if (!img || !imageContainer) return;
            
            const containerRect = imageContainer.getBoundingClientRect();
            const imgRect = img.getBoundingClientRect();
            
            const scaleX = containerRect.width / img.naturalWidth;
            const scaleY = containerRect.height / img.naturalHeight;
            currentScale = Math.min(scaleX, scaleY, 1);
            
            currentPanX = (containerRect.width - img.naturalWidth * currentScale) / 2;
            currentPanY = (containerRect.height - img.naturalHeight * currentScale) / 2;
            
            updateImageTransform();
            updateStatus('zoom-status', `${Math.round(currentScale * 100)}%`);
            console.log('Image fitted to container');
        }

function setupModeToggleRobust() {
    console.log("[DEBUG] Starting robust toggle setup");
    
    const toggle = document.getElementById('mode-toggle-switch');
    if (!toggle) {
        console.error("[DEBUG] Mode toggle switch element not found!");
        return;
    }
    
    console.log("[DEBUG] Toggle element found:", toggle);
    
    // Wait longer and try multiple times if needed
    setTimeout(() => {
        console.log("[DEBUG] Setting up toggle after delay");
        
        // Check user roles
        if (isLabeller() && isReviewer()) {
            toggle.checked = currentMode === 'validate';
            toggle.disabled = false;
            console.log("[DEBUG] Toggle enabled for dual-role user, checked:", toggle.checked);
        } else {
            toggle.disabled = true;
            console.log("[DEBUG] Toggle disabled - user roles:", { labeller: isLabeller(), reviewer: isReviewer() });
            return; // Don't set up event listeners if disabled
        }
        
        // Remove any existing event listeners by cloning
        const newToggle = toggle.cloneNode(true);
        toggle.parentNode.replaceChild(newToggle, toggle);
        console.log("[DEBUG] Toggle element cloned and replaced");
        
        // Set up event listeners on the new element
        setupRobustToggleEventListeners(newToggle);
        addToggleVisualFeedback(newToggle);
        
        // Initial styling update
        updateLabelStyling(newToggle.checked);
        
    }, 200); // Increased delay
}

function setupRobustToggleEventListeners(toggle) {
    console.log("[DEBUG] Setting up robust toggle event listeners");
    
    // Main change event with detailed logging
    toggle.addEventListener('change', function(e) {
        console.log("[DEBUG] Toggle changed - checked:", this.checked, "disabled:", this.disabled);
        if (!this.disabled) {
            handleModeToggle.call(this, e);
        } else {
            console.log("[DEBUG] Toggle change ignored - element is disabled");
        }
    });
    
    // Find toggle container with multiple fallback methods
    let toggleContainer = null;
    
    // Method 1: Look for parent with toggle-switch class
    toggleContainer = toggle.closest('.toggle-switch');
    
    // Method 2: Look for next sibling div
    if (!toggleContainer) {
        toggleContainer = toggle.nextElementSibling;
    }
    
    // Method 3: Look for parent div
    if (!toggleContainer) {
        toggleContainer = toggle.parentElement;
    }
    
    console.log("[DEBUG] Toggle container found:", toggleContainer);
    
    // Set up container clicking if found
    if (toggleContainer) {
        toggleContainer.addEventListener('click', function(e) {
            console.log("[DEBUG] Toggle container clicked");
            e.preventDefault();
            e.stopPropagation();
            
            if (isLabeller() && isReviewer() && !toggle.disabled) {
                console.log("[DEBUG] Toggling via container click");
                toggle.checked = !toggle.checked;
                const changeEvent = new Event('change', { bubbles: true });
                toggle.dispatchEvent(changeEvent);
            } else {
                console.log("[DEBUG] Container click ignored - insufficient permissions or disabled");
            }
        });
        
        // Prevent toggle from being clicked twice
        toggle.addEventListener('click', function(e) {
            e.stopPropagation();
        });
    }
    
    // Label clicking with improved selectors
    const labelLeft = document.getElementById('mode-label-left');
    const labelRight = document.getElementById('mode-label-right');
    
    console.log("[DEBUG] Labels found - Left:", labelLeft, "Right:", labelRight);
    
    if (labelLeft) {
        labelLeft.addEventListener('click', function(e) {
            console.log("[DEBUG] Left label clicked (Label mode)");
            e.preventDefault();
            e.stopPropagation();
            
            if (isLabeller() && isReviewer() && !toggle.disabled) {
                console.log("[DEBUG] Setting to label mode");
                toggle.checked = false;
                const changeEvent = new Event('change', { bubbles: true });
                toggle.dispatchEvent(changeEvent);
            } else {
                console.log("[DEBUG] Left label click ignored");
            }
        });
        
        // Add cursor pointer
        labelLeft.style.cursor = 'pointer';
    }
    
    if (labelRight) {
        labelRight.addEventListener('click', function(e) {
            console.log("[DEBUG] Right label clicked (Validate mode)");
            e.preventDefault();
            e.stopPropagation();
            
            if (isLabeller() && isReviewer() && !toggle.disabled) {
                console.log("[DEBUG] Setting to validate mode");
                toggle.checked = true;
                const changeEvent = new Event('change', { bubbles: true });
                toggle.dispatchEvent(changeEvent);
            } else {
                console.log("[DEBUG] Right label click ignored");
            }
        });
        
        // Add cursor pointer
        labelRight.style.cursor = 'pointer';
    }
    
    toggle._hasToggleListeners = true;
    console.log("[DEBUG] Toggle event listeners setup complete");
}

function addToggleVisualFeedback(toggle) {
    console.log("[DEBUG] Adding visual feedback to toggle");
    
    // Find the toggle container using multiple methods
    let toggleContainer = toggle.closest('.toggle-switch') || 
                         toggle.nextElementSibling || 
                         toggle.parentElement;
    
    const labelLeft = document.getElementById('mode-label-left');
    const labelRight = document.getElementById('mode-label-right');
    
    if (toggleContainer) {
        console.log("[DEBUG] Adding hover effects to toggle container");
        
        // Hover effects for toggle container
        toggleContainer.addEventListener('mouseenter', function() {
            if (!toggle.disabled) {
                this.style.boxShadow = '0 0 0 3px rgba(59, 130, 246, 0.3)';
                this.style.transition = 'box-shadow 0.2s ease';
            }
        });
        
        toggleContainer.addEventListener('mouseleave', function() {
            this.style.boxShadow = '';
        });
        
        // Click feedback
        toggleContainer.addEventListener('mousedown', function() {
            if (!toggle.disabled) {
                this.style.transform = 'scale(0.95)';
                this.style.transition = 'transform 0.1s ease';
            }
        });
        
        toggleContainer.addEventListener('mouseup', function() {
            this.style.transform = '';
        });
    }
    
    // Label hover effects
    [labelLeft, labelRight].forEach((label, index) => {
        if (label) {
            console.log(`[DEBUG] Adding hover effects to label ${index === 0 ? 'left' : 'right'}`);
            
            label.addEventListener('mouseenter', function() {
                if (isLabeller() && isReviewer() && !toggle.disabled) {
                    this.style.backgroundColor = 'rgba(59, 130, 246, 0.1)';
                    this.style.borderRadius = '4px';
                    this.style.padding = '2px 4px';
                    this.style.transition = 'background-color 0.2s ease';
                }
            });
            
            label.addEventListener('mouseleave', function() {
                this.style.backgroundColor = '';
                this.style.borderRadius = '';
                this.style.padding = '';
            });
            
            // Click feedback
            label.addEventListener('mousedown', function() {
                if (isLabeller() && isReviewer() && !toggle.disabled) {
                    this.style.transform = 'scale(0.95)';
                    this.style.transition = 'transform 0.1s ease';
                }
            });
            
            label.addEventListener('mouseup', function() {
                this.style.transform = '';
            });
        }
    });
}

function handleModeToggle(e) {
    console.log("[DEBUG] handleModeToggle called, checked:", this.checked);
    
    if (!(isLabeller() && isReviewer())) {
        console.log("[DEBUG] User doesn't have both roles, toggle disabled");
        this.checked = currentMode === 'validate'; // Revert to current state
        return;
    }
    
    if (this.disabled) {
        console.log("[DEBUG] Toggle is disabled, ignoring change");
        return;
    }
    
    const isValidateMode = this.checked;
    const oldMode = currentMode;
    currentMode = isValidateMode ? 'validate' : 'label';
    
    console.log(`[DEBUG] Mode toggle: ${oldMode} -> ${currentMode}`);
    
    // Update label styling
    updateLabelStyling(isValidateMode);
    
    // Update UI for the new mode
    updateUIForMode(isValidateMode);
    
    // Sync with main.js if available
    if (window.AppState) {
        window.AppState.currentMode = currentMode;
        console.log("[DEBUG] Updated AppState.currentMode to:", currentMode);
    }
    
    // Call main.js toggle function if available
    if (typeof window.toggleValidationMode === 'function') {
        console.log("[DEBUG] Calling window.toggleValidationMode");
        window.toggleValidationMode(isValidateMode);
    } else {
        console.log("[DEBUG] window.toggleValidationMode not available");
    }
    
    // Show user feedback
    showMessage(`Switched to ${currentMode} mode`, 'info');
    
    console.log(`[DEBUG] Mode successfully changed to: ${currentMode}`);
}

function updateLabelStyling(isValidateMode) {
    console.log("[DEBUG] Updating label styling for mode:", isValidateMode ? 'validate' : 'label');
    
    const labelLeft = document.getElementById('mode-label-left');
    const labelRight = document.getElementById('mode-label-right');
    
    if (labelLeft && labelRight) {
        // Reset all classes first
        labelLeft.classList.remove('text-blue-600', 'text-blue-500', 'text-gray-500');
        labelRight.classList.remove('text-blue-600', 'text-blue-500', 'text-gray-500');
        
        if (isValidateMode) {
            // Validate mode (right side active)
            labelLeft.style.fontWeight = 'normal';
            labelLeft.classList.add('text-gray-500');
            
            labelRight.style.fontWeight = 'bold';
            labelRight.classList.add('text-blue-600');
            
            console.log("[DEBUG] Styled for validate mode - left gray, right blue");
        } else {
            // Label mode (left side active)
            labelLeft.style.fontWeight = 'bold';
            labelLeft.classList.add('text-blue-600');
            
            labelRight.style.fontWeight = 'normal';
            labelRight.classList.add('text-gray-500');
            
            console.log("[DEBUG] Styled for label mode - left blue, right gray");
        }
    } else {
        console.error("[DEBUG] Label elements not found for styling");
    }
}

        // Role management
        async function initializeUserRoles() {
            try {
                console.log('🔍 Fetching user info...');
                const response = await fetch('/api/users/me');
                if (!response.ok) throw new Error('Failed to fetch user info');
                const userData = await response.json();
                userRoles = userData.roles || [];
                console.log('👤 User roles:', userRoles);
                await setupBasedOnRole();
            } catch (error) {
                console.error('❌ Error initializing user:', error);
            }
        }
 
        function isAdministrator() {
            const hasAdminRole = userRoles.includes('Administrator');
            const hasMainJsAdminRole = window.AppState?.user?.role === 'Default Admin';
            return hasAdminRole || hasMainJsAdminRole;
        }
 
        function isLabeller() {
            const hasLabellerRole = userRoles.includes('Labeller');
            const hasMainJsAdminRole = window.AppState?.user?.role === 'Default Admin';
            return hasLabellerRole || isAdministrator() || hasMainJsAdminRole;
        }
 
        function isReviewer() {
            const hasReviewerRole = userRoles.includes('Reviewer');
            const hasMainJsReviewerRole = window.AppState?.user?.role === 'Reviewer';
            const hasMainJsAdminRole = window.AppState?.user?.role === 'Default Admin';
            return hasReviewerRole || isAdministrator() || hasMainJsReviewerRole || hasMainJsAdminRole;
        }

            console.log('isAdministrator:', isAdministrator());

            }

                currentMode = 'validate';
                if (modeToggleSwitch) {
                    modeToggleSwitch.checked = true;
                    modeToggleSwitch.disabled = true;
                }

                }

                // Pure Labeller

                currentMode = 'label';
                if (modeToggleSwitch) {
                    modeToggleSwitch.checked = false;
                    modeToggleSwitch.disabled = true;

                currentMode = 'label';

        // Drawing Annotations (simplified since main.js handles most annotation logic)
        function drawAnnotations() {
            if (!annotationCanvas || !ctx) return;
           
            clearCanvas();
            if (!annotations || !annotations.length) return;
           
            annotations.forEach(annotation => {
                drawAnnotation(annotation);
            });
        }
 
        function drawAnnotation(annotation) {
            if (!ctx) return;
            
            let points = [];
            
            if (annotation.segmentation && annotation.segmentation.length > 0) {
                const segmentation = annotation.segmentation[0];
                for (let i = 0; i < segmentation.length; i += 2) {
                    points.push({ x: segmentation[i], y: segmentation[i + 1] });
                }
            } else if (annotation.points && Array.isArray(annotation.points)) {
                points = annotation.points;
            } else {
                console.warn('Annotation has no valid points data:', annotation);
                return;
            }
            
            if (points.length < 3) {
                console.warn('Annotation has insufficient points:', points.length);
                return;
            }
           
            const categoryId = annotation.category_id || annotation.categoryId;
            const category = categories.find(c => c.id === categoryId || c.name === annotation.categoryName);
            const color = category?.color || '#FF0000';
           
            ctx.strokeStyle = color;
            ctx.lineWidth = 2;
            ctx.fillStyle = `${color}33`;
           
            ctx.beginPath();
            const firstPoint = normalizePoint(points[0]);
            ctx.moveTo(firstPoint.x, firstPoint.y);
           
            for (let i = 1; i < points.length; i++) {
                const point = normalizePoint(points[i]);
                ctx.lineTo(point.x, point.y);
            }
           
            ctx.closePath();
            ctx.stroke();
            ctx.fill();
        }
 
        function normalizePoint(point) {
            const img = document.getElementById('main-image');
            if (!img) return { x: 0, y: 0 };
           
            return {
                x: (point.x / img.naturalWidth) * annotationCanvas.width,
                y: (point.y / img.naturalHeight) * annotationCanvas.height
            };
        }

        }

        async function deleteAnnotationsAlternative() {
            const batchId = window.AppState?.currentBatch || currentBatchId;
            const imageId = window.AppState?.currentImageId || currentImageId;
            
            if (!batchId || !imageId) {
                throw new Error('No batch or image selected');
            }
            
            try {
                const response = await fetch('/api/annotations/clear', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        batch_id: batchId,
                        image_id: imageId
                    })
                });
                
                if (!response.ok) {
                    throw new Error(`Clear endpoint failed: ${response.statusText}`);
                }
                
                console.log('Successfully cleared annotations via clear endpoint');
            } catch (clearError) {
                console.warn('Clear endpoint not available, trying DELETE method:', clearError);
                
                const deleteResponse = await fetch(`/api/annotations/${batchId}/${imageId}?batch_id=${batchId}&image_id=${imageId}`, {
                    method: 'DELETE',
                    headers: {
                        'Content-Type': 'application/json',
                    }
                });
                
                if (!deleteResponse.ok) {
                    throw new Error(`DELETE request failed: ${deleteResponse.statusText}`);
                }
                
                console.log('Successfully deleted annotations via DELETE method');
            }
        }

        // Setup image pan and zoom functionality
        function setupImagePanZoom() {
            if (!imageContainer) return;
            
            imageContainer.addEventListener('dragstart', (e) => e.preventDefault());
            imageContainer.addEventListener('mousedown', handlePanStart);
            document.addEventListener('mousemove', handlePanMove);
            document.addEventListener('mouseup', handlePanEnd);
            imageContainer.addEventListener('wheel', handleZoom);
            imageContainer.addEventListener('imageLoaded', resetPanZoom);
            
            console.log('Image pan/zoom functionality initialized');
        }
        
        function handlePanStart(e) {
            if (e.shiftKey && e.button === 0) {
                e.preventDefault();
                isPanning = true;
                panStartX = e.clientX - currentPanX;
                panStartY = e.clientY - currentPanY;
                imageContainer.classList.add('panning');
                console.log('Pan started');
            }
        }
        
        function handlePanMove(e) {
            if (!isPanning) {
                if (e.shiftKey && imageContainer) {
                    imageContainer.style.cursor = 'grab';
                } else if (imageContainer) {
                    imageContainer.style.cursor = '';
                }
                return;
            }
            
            e.preventDefault();
            currentPanX = e.clientX - panStartX;
            currentPanY = e.clientY - panStartY;
            
            updateImageTransform();
        }
        
        function handlePanEnd(e) {
            if (isPanning) {
                isPanning = false;
                imageContainer.classList.remove('panning');
                imageContainer.style.cursor = '';
                console.log('Pan ended');
            }
        }
        
        function handleZoom(e) {
            if (e.ctrlKey) {
                e.preventDefault();
                
                const zoomIntensity = 0.1;
                const wheel = e.deltaY < 0 ? 1 : -1;
                const zoom = Math.exp(wheel * zoomIntensity);
                
                const rect = imageContainer.getBoundingClientRect();
                const centerX = e.clientX - rect.left;
                const centerY = e.clientY - rect.top;
                
                const newScale = Math.min(Math.max(0.1, currentScale * zoom), 5);
                
                if (newScale !== currentScale) {
                    currentPanX = centerX - (centerX - currentPanX) * (newScale / currentScale);
                    currentPanY = centerY - (centerY - currentPanY) * (newScale / currentScale);
                    currentScale = newScale;
                    
                    updateImageTransform();
                    updateStatus('zoom-status', `${Math.round(currentScale * 100)}%`);
                }
            }
        }
        
        function updateImageTransform() {
            const canvasWrapper = document.getElementById('canvas-wrapper');
            if (canvasWrapper) {
                canvasWrapper.style.transform = `translate(${currentPanX}px, ${currentPanY}px) scale(${currentScale})`;
                canvasWrapper.style.transformOrigin = '0 0';
                canvasWrapper.style.transition = isPanning ? 'none' : 'transform 0.1s ease-out';
            }
        }
